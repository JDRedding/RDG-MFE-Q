def induce_partition(X, threshold=0.1):
    """Simple clustering: group modes if |Xi - Xj| < threshold"""
    sorted_idx = np.argsort(X)
    sorted_X = X[sorted_idx]
    blocks = []
    current_block = [sorted_idx[0]]
    for i in range(1, len(X)):
        if abs(sorted_X[i] - sorted_X[i-1]) < threshold:
            current_block.append(sorted_idx[i])
        else:
            blocks.append(current_block)
            current_block = [sorted_idx[i]]
    blocks.append(current_block)
    return [sorted(b) for b in blocks], len(blocks)

# Function to check A1-A3
def check_A1_A3(blocks):
    k = len(blocks)
    if not (1 < k < 7):
        return False
    sizes = [len(b) for b in blocks]
    if 3 not in sizes:
        return False  # A2
    if any(s > 3 for s in sizes):
        return False  # A3
    return True
